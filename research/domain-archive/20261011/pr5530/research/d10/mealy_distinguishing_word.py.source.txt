#!/usr/bin/env python3
"""Дослід D10: найкоротше слово, що розрізняє два автомати Мілі.

Це незалежний інструмент дослідження, НЕ команда/координата D10.
"""
from __future__ import annotations

from collections import deque
from itertools import product
import unittest


def перевірити(автомат: tuple, абетка: int) -> None:
    if type(абетка) is not int or not 1 <= абетка <= 4:
        raise ValueError("алфавіт мусить мати від 1 до 4 символів у тестовому оракулі")
    if not isinstance(автомат, tuple) or not автомат:
        raise ValueError("скінченний непорожній набір станів обов'язковий")
    for рядок in автомат:
        if not isinstance(рядок, tuple) or len(рядок) != абетка:
            raise ValueError("автомат має бути повним і детермінованим")
        for перехід in рядок:
            if not isinstance(перехід, tuple) or len(перехід) != 2:
                raise ValueError("перехід має бути парою (стан, біт виходу)")
            новий, вихід = перехід
            if type(новий) is not int or not 0 <= новий < len(автомат):
                raise ValueError("новий стан поза множиною станів")
            if type(вихід) is not int or вихід not in (0, 1):
                raise ValueError("вихід у цьому оракулі — точний біт 0/1")


def найкоротше_розрізнення(лівий: tuple, правий: tuple, абетка: int) -> tuple[int, ...] | None:
    """None — еквівалентні; інакше найкоротше лексикографічно перше слово.

    Початковий стан обох автоматів — 0. Пошук BFS над парами станів.
    """
    перевірити(лівий, абетка)
    перевірити(правий, абетка)
    черга = deque([(0, 0, ())])
    відвідано = {(0, 0)}
    while черга:
        стан_л, стан_п, слово = черга.popleft()
        for символ in range(абетка):
            наступний_л, вихід_л = лівий[стан_л][символ]
            наступний_п, вихід_п = правий[стан_п][символ]
            нове_слово = слово + (символ,)
            if вихід_л != вихід_п:
                return нове_слово
            пара = (наступний_л, наступний_п)
            if пара not in відвідано:
                відвідано.add(пара)
                черга.append((*пара, нове_слово))
    return None


def спостереження(автомат: tuple, слово: tuple[int, ...]) -> tuple[int, ...]:
    стан = 0
    виходи = []
    for символ in слово:
        стан, вихід = автомат[стан][символ]
        виходи.append(вихід)
    return tuple(виходи)


def незалежний_оракул(лівий: tuple, правий: tuple, абетка: int) -> tuple[int, ...] | None:
    """Повний перебір усіх слів до |Q1|×|Q2|, НЕ BFS пар станів."""
    перевірити(лівий, абетка)
    перевірити(правий, абетка)
    for довжина in range(1, len(лівий) * len(правий) + 1):
        for слово in product(range(абетка), repeat=довжина):
            if спостереження(лівий, слово) != спостереження(правий, слово):
                return слово
    return None


class ДоказовіПеревірки(unittest.TestCase):
    def test_свідки(self) -> None:
        а = (((1, 0), (0, 0)), ((1, 1), (1, 0)))
        б = (((1, 0), (0, 0)), ((1, 0), (1, 0)))
        self.assertEqual(найкоротше_розрізнення(а, б, 2), (0, 0))
        self.assertNotEqual(спостереження(а, (0, 0)), спостереження(б, (0, 0)))
        self.assertIsNone(найкоротше_розрізнення(а, а, 2))
        недосяжний = (((0, 0), (0, 0)), ((1, 1), (1, 1)))
        еквівалентний = (((0, 0), (0, 0)), ((1, 0), (1, 0)))
        self.assertIsNone(найкоротше_розрізнення(недосяжний, еквівалентний, 2))

    def test_негативні_випадки(self) -> None:
        for автомат, абетка in [
            ((), 2),
            ((((0, 0),),), 2),
            ((((-1, 0), (0, 0)),), 2),
            ((((0, 0), (0, 2)),), 2),
        ]:
            with self.subTest(автомат=автомат, абетка=абетка):
                with self.assertRaises(ValueError):
                    найкоротше_розрізнення(автомат, (((0, 0), (0, 0)),), абетка)

    def test_повний_двостанний_простір(self) -> None:
        # 4^(2×2)=256 автоматів; порівнюємо всі 256²=65536 впорядкованих пар.
        елементи = tuple(product(range(2), repeat=2))
        автомати = []
        for клітинки in product(елементи, repeat=4):
            автомати.append((клітинки[:2], клітинки[2:]))
        self.assertEqual(len(автомати), 256)
        for а in автомати:
            for б in автомати:
                self.assertEqual(
                    найкоротше_розрізнення(а, б, 2),
                    незалежний_оракул(а, б, 2),
                )


if __name__ == "__main__":
    unittest.main(verbosity=2)
