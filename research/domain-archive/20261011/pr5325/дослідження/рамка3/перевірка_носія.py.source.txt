"""Відтворювані позитивні й негативні свідки для дослідної рамки."""
import random
import sys
import unittest
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent))
import носій as р


class ПеревіркаРамки(unittest.TestCase):
    def test_відомі_приклади(self):
        зразки = [
            ("10 01", "00", 1),
            ("10 001 01", "08", 2),
            ("10 001 00 000 01", "23d0", 4),
            ("10 100 00 10 111 00 1 00 0 01 01", "2109895eb5", 7),
            ("10 001 00 10 000 01 01", "3a38ed", 5),
        ]
        for текст, очікуване, байтів_т5 in зразки:
            with self.subTest(текст=текст):
                слова = текст.split()
                фізичні = р.упакувати(слова)
                self.assertEqual(фізичні.hex(), очікуване)
                self.assertEqual(р.розпакувати(фізичні), слова)
                self.assertEqual(р.байтів_т5(слова), байтів_т5)
                self.assertLessEqual(len(фізичні), байтів_т5)

    def test_місткість_до_трьох_байтів(self):
        self.assertEqual(р.групи()[:3],
            ((1, 5, 11, 139), (2, 12, 17, 57504), (3, 18, 22, 8716160)))
        for байтів, _перша, _остання, обсяг in р.групи():
            self.assertLessEqual(обсяг, 1 << (8 * байтів))

    def test_ранжування_незалежних_послідовностей(self):
        випадкові = random.Random(20261010)
        перевірок = 0
        for _, перша, остання, _ in р.групи():
            for довжина in range(перша, остання + 1):
                кількість = р.місткість(довжина)
                if кількість == 0:
                    continue  # Немає жодного допустимого слова цієї довжини.
                варіанти = {0, кількість // 2, кількість - 1}
                варіанти.update(випадкові.randrange(кількість) for _ in range(50))
                for число in варіанти:
                    трити = р.за_рангом(довжина, число)
                    self.assertEqual(р.ранг(трити), число)
                    self.assertNotIn((2, 2), list(zip(трити, трити[1:])))
                    перевірок += 1
        self.assertGreaterEqual(перевірок, 2000)

    def test_помилкові_рамки(self):
        for текст in ("10 001", "10 01 10 01", "10 001 01 01",
                      "01 10", "10 0000000000 01", "10 01 0"):
            with self.subTest(текст=текст), self.assertRaises(ValueError):
                р.упакувати(текст.split())

    def test_незайняті_коди(self):
        for дані in (b"\xff", b"\xff\xff", b"\xff\xff\xff", b""):
            with self.subTest(дані=дані), self.assertRaises(ValueError):
                р.розпакувати(дані)

    def test_послідовності_вкладених_рамок(self):
        випадкові = random.Random(1042026)
        for _ in range(1000):
            # Одна зовнішня форма, не більше трьох вкладених рамок.
            слова = ["10"]
            for _ in range(випадкові.randrange(4)):
                слова += ["00", "10", "00", "001", "01"]
            слова += ["00", "111", "01"]
            try:
                фізичні = р.упакувати(слова)
            except ValueError as e:
                if "межу дослідного носія" in str(e):
                    continue
                raise
            self.assertEqual(р.розпакувати(фізичні), слова)


    def test_вичерпно_усі_трити_до_12(self):
        """797161 рядок незалежного алфавітного перебору; ніякого випадкового семплювання.

        Доказ обмежений синтаксисом поточного рамкового носія, НЕ
        семантичною еквівалентністю канонічному D2-оракулу або T5.
        """
        from itertools import product

        очікувана_рамка = (0, 0, 0, 0, 0, 1, 0, 2, 4, 12, 32, 88, 240)
        очікувана_форма = (0, 0, 0, 0, 0, 1, 0, 2, 2, 12, 24, 77, 184)
        перевірено = 0
        рамок = 0
        форм = 0

        def незалежна_форма(слова):
            if len(слова) < 2 or слова[0] != "10" or слова[-1] != "01":
                return False
            глибина = 0
            for номер, слово in enumerate(слова):
                if not 1 <= len(слово) <= 9 or any(біт not in "01" for біт in слово):
                    return False
                if слово == "10":
                    глибина += 1
                elif слово == "01":
                    глибина -= 1
                    if глибина < 0 or (глибина == 0 and номер != len(слова) - 1):
                        return False
                elif глибина <= 0:
                    return False
            return глибина == 0

        for довжина in range(13):
            кандидати = []
            для_форми = 0
            for цифри in product("012", repeat=довжина):
                рядок = "".join(цифри)
                перевірено += 1
                рамка = (
                    (довжина == 5 and рядок == "10201")
                    or (довжина >= 7 and рядок.startswith("102")
                        and рядок.endswith("201") and "22" not in рядок)
                )
                if рамка:
                    кандидати.append(рядок)
                else:
                    try:
                        р.ранг(tuple(map(int, рядок)))
                    except ValueError:
                        pass
                    else:
                        self.fail("Ранг прийняв сторонній тритовий рядок: " + рядок)

            self.assertEqual(len(кандидати), очікувана_рамка[довжина])
            self.assertEqual(р.місткість(довжина), len(кандидати))
            for порядковий, рядок in enumerate(кандидати):
                трити = tuple(map(int, рядок))
                self.assertEqual(р.ранг(трити), порядковий)
                self.assertEqual(р.за_рангом(довжина, порядковий), трити)
                слова = рядок.split("2")
                допущено = незалежна_форма(слова)
                байти = 1 if довжина <= 11 else 2
                зміщення = (sum(очікувана_рамка[5:довжина])
                           if довжина <= 11 else 0)
                фізичні = (зміщення + порядковий).to_bytes(байти, "big")
                if допущено:
                    для_форми += 1
                    self.assertEqual(р.трити_форми(слова), трити)
                    self.assertEqual(р.упакувати(слова), фізичні)
                    self.assertEqual(р.розпакувати(фізичні), слова)
                else:
                    with self.assertRaises(ValueError):
                        р.упакувати(слова)
                    with self.assertRaises(ValueError):
                        р.розпакувати(фізичні)
            self.assertEqual(для_форми, очікувана_форма[довжина])
            рамок += len(кандидати)
            форм += для_форми

        self.assertEqual((перевірено, рамок, форм), (797161, 379, 302))
        self.assertEqual(рамок - форм, 77)

        # Додаткова повна перевірка всіх 256 однобайтових кодів.
        допущені_байти = 0
        for число in range(256):
            дані = bytes([число])
            try:
                слова = р.розпакувати(дані)
            except ValueError:
                continue
            self.assertEqual(р.упакувати(слова), дані)
            допущені_байти += 1
        self.assertEqual(допущені_байти, 118)

if __name__ == "__main__":
    unittest.main()
